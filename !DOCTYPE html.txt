        function getVeredaById(id) {
            return veredasData.find(v => v.id === id);
        }

        async function callGeminiAPI(prompt, resultContainer, button) {
            button.disabled = true;
            resultContainer.innerHTML = '<p class="text-center text-gray-500">✨ Generando respuesta...</p>';
            resultContainer.classList.remove('hidden');

            const apiKey = "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
            };

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    throw new Error(`Error de API: ${response.statusText}`);
                }

                const result = await response.json();
                const text = result.candidates?.[0]?.content?.parts?.[0]?.text;

                if (text) {
                    resultContainer.innerHTML = `<div class="prose max-w-none">${text.replace(/\n/g, '<br>')}</div><button onclick="this.parentElement.classList.add('hidden')" class="mt-2 text-sm text-red-500 font-semibold">Cerrar</button>`;
                } else {
                    throw new Error("No se recibió contenido válido de la API.");
                }
            } catch (error) {
                console.error("Error llamando a la API de Gemini:", error);
                resultContainer.innerHTML = `<p class="text-red-600">Lo sentimos, no se pudo generar la respuesta. Por favor, inténtelo de nuevo.</p><button onclick="this.parentElement.classList.add('hidden')" class="mt-2 text-sm text-red-500 font-semibold">Cerrar</button>`;
            } finally {
                button.disabled = false;
            }
        }

        function generateTastingProfile(button, veredaId) {
            const vereda = getVeredaById(veredaId);
            if (!vereda) return;

            const scores = Object.entries(vereda.perfil).map(([key, value]) => `${key}: ${value}`).join(', ');
            const prompt = `Eres un catador de café de clase mundial. Basado en el siguiente perfil sensorial para un café de la vereda de ${vereda.nombre} en Ituango, Colombia, escribe una descripción de cata evocadora y atractiva de unas 60-80 palabras para un comprador internacional. Notas clave: ${vereda.notas}. Puntajes sensoriales (sobre 10): ${scores}.`;
            
            const resultContainer = document.getElementById(`gemini-result-${veredaId}`);
            callGeminiAPI(prompt, resultContainer, button);
        }

        function generateFoodPairing(button, veredaId) {
            const vereda = getVeredaById(veredaId);
            if (!vereda) return;
            
            const prompt = `Como chef experto y conocedor de café, sugiere tres maridajes (food pairings) específicos y creativos que complementarían un café con estas características: ${vereda.notas}. Describe por qué cada maridaje funciona bien. Sé breve y atractivo para un menú de una cafetería de especialidad.`;

            const resultContainer = document.getElementById(`gemini-result-${veredaId}`);
            callGeminiAPI(prompt, resultContainer, button);
        }

        function createRadarChart(canvasId, data) {
            const ctx = document.getElementById(canvasId).getContext('2d');
            const labels = Object.keys(data);
            const values = Object.values(data);

            new Chart(ctx, {
                type: 'radar',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Puntaje Sensorial',
                        data: values,
                        backgroundColor: 'rgba(6, 214, 160, 0.2)',
                        borderColor: 'rgba(6, 214, 160, 1)',
                        borderWidth: 2,
                        pointBackgroundColor: 'rgba(6, 214, 160, 1)',
                        pointBorderColor: '#fff',
                        pointHoverBackgroundColor: '#fff',
                        pointHoverBorderColor: 'rgba(6, 214, 160, 1)'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        r: {
                            angleLines: { color: 'rgba(7, 59, 76, 0.2)' },
                            grid: { color: 'rgba(7, 59, 76, 0.2)' },
                            pointLabels: { font: { size: 12, weight: '600' }, color: '#073B4C' },
                            ticks: { beginAtZero: true, min: 7, max: 10, stepSize: 1, backdropColor: 'rgba(255, 255, 255, 0.75)', color: '#4A5568' }
                        }
                    },
                    plugins: {
                        legend: { display: false },
                        tooltip: { callbacks: { title: function(tooltipItems) { const item = tooltipItems[0]; let label = item.chart.data.labels[item.dataIndex]; if (Array.isArray(label)) { return label.join(' '); } else { return label; } } } }
                    }
                }
            });
        }
